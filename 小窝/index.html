  <script>

    // ========== 配置 ==========

    const GOAL = 5000;              // 目标金额

    const MONTHLY_BUDGET = 2370;    // 每月可花

    const DAILY_BUDGET = 79;        // 每日额度

    const STAGES = [

      { target: 500, name: '打地基', boss: '连续7天不超支' },

      { target: 1500, name: '砌墙', boss: '这周不点外卖' },

      { target: 3000, name: '盖屋顶', boss: '连续14天存钱' },

      { target: 5000, name: '完工', boss: '无' }

    ];


    // ========== 数据层 ==========

    const store = {

      get(key, fallback) {

        const v = localStorage.getItem('xiaowo_' + key);

        return v === null ? fallback : JSON.parse(v);

      },

      set(key, value) {

        localStorage.setItem('xiaowo_' + key, JSON.stringify(value));

      }

    };


    let state = {

      saved: store.get('saved', 0),

      streak: store.get('streak', 0),

      lastSaveDate: store.get('lastSaveDate', null),

      todaySpent: store.get('todaySpent', 0),

      todayDate: store.get('todayDate', new Date().toDateString()),

      neighborSaved: store.get('neighborSaved', true)

    };


    // 新的一天，重置今日花费

    const today = new Date().toDateString();

    if (state.todayDate !== today) {

      state.todaySpent = 0;

      state.todayDate = today;

      store.set('todaySpent', 0);

      store.set('todayDate', today);

    }


    // ========== 渲染 ==========

    function render() {

      // 今日还能花

      const remaining = Math.max(0, DAILY_BUDGET - state.todaySpent);

      const amountEl = document.getElementById('todayAmount');

      amountEl.textContent = '¥' + remaining;

      amountEl.classList.toggle('warning', remaining < 30);


      // 进度

      const percent = Math.min(100, (state.saved / GOAL) * 100);

      document.getElementById('progressFill').style.width = percent + '%';

      document.getElementById('savedLabel').textContent = '已存 ¥' + state.saved;

      document.getElementById('goalLabel').textContent = '目标 ¥' + GOAL;


      // 房子标签

      const label = document.getElementById('myHouseLabel');

      let stageText = '';

      if (percent < 10) stageText = '🧱 打地基';

      else if (percent < 30) stageText = '🧱 砌墙中';

      else if (percent < 60) stageText = '🏠 盖屋顶';

      else if (percent < 90) stageText = '🪟 装窗户';

      else if (percent < 100) stageText = '🛋️ 布置家';

      else stageText = '✨ 完工亮灯';

      label.textContent = stageText + ' · ' + Math.round(percent) + '%';


      // streak

      document.getElementById('streakDays').textContent = state.streak;


      // 邻居状态

      const neighborEl = document.getElementById('neighborStatus');

      if (state.neighborSaved) {

        neighborEl.textContent = '✅ 今日已存';

        neighborEl.classList.remove('missed');

      } else {

        neighborEl.textContent = '❌ 今日未存';

        neighborEl.classList.add('missed');

      }


      // 当前阶段

      let currentStage = 0;

      for (let i = 0; i < STAGES.length; i++) {

        if (state.saved < STAGES[i].target) {

          currentStage = i;

          break;

        }

        currentStage = i;

      }

      const stage = STAGES[currentStage];

      document.getElementById('stageBadge').textContent = `第${currentStage + 1}关 / 共${STAGES.length}关`;

      document.getElementById('bossName').textContent = stage.boss;


      // 成就墙

      const memoryTitle = document.getElementById('memoryTitle');

      const memoryDesc = document.getElementById('memoryDesc');

      if (state.saved >= 10000) {

        memoryTitle.textContent = '书房的台灯';

        memoryDesc.textContent = '你超过了 70% 的同龄人';

      } else if (state.saved >= 5000) {

        memoryTitle.textContent = '客厅的沙发';

        memoryDesc.textContent = '你可以裸辞 1 个月';

      } else if (state.saved >= 1000) {

        memoryTitle.textContent = '门口的信箱';

        memoryDesc.textContent = '你拥有了第一笔应急金 · 1000元';

      } else {

        memoryTitle.textContent = '门口的信箱';

        memoryDesc.textContent = '存到 1000 元就能解锁';

      }

    }


    // ========== 动作 ==========

    function saveMoney(amount) {

      state.saved += amount;

      state.todaySpent += amount;

      store.set('saved', state.saved);

      store.set('todaySpent', state.todaySpent);


      // 更新 streak

      const todayStr = new Date().toDateString();

      if (state.lastSaveDate !== todayStr) {

        state.streak += 1;

        state.lastSaveDate = todayStr;

        store.set('streak', state.streak);

        store.set('lastSaveDate', todayStr);

      }


      render();

    }


    function recordSpend(amount) {

      state.todaySpent += amount;

      store.set('todaySpent', state.todaySpent);

      render();

    }


    // ========== 事件绑定 ==========

    document.getElementById('saveBtn').addEventListener('click', () => {

      saveMoney(10);

    });


    document.getElementById('recordBtn').addEventListener('click', () => {

      recordSpend(23);

    });


    // 后悔药选项

    document.querySelectorAll('.remedy-option').forEach((el) => {

      el.addEventListener('click', () => {

        const type = el.dataset.remedy;

        if (type === 'A') {

          state.todaySpent = Math.max(0, state.todaySpent - 41);

        } else if (type === 'B') {

          state.saved = Math.max(0, state.saved - 41);

          state.streak = 0;

          store.set('streak', 0);

        } else if (type === 'C') {

          state.todaySpent = Math.max(0, state.todaySpent - 20);

        }

        store.set('saved', state.saved);

        store.set('todaySpent', state.todaySpent);

        render();

      });

    });


    // ========== 时钟 ==========

    function updateClock() {

      const now = new Date();

      const h = String(now.getHours()).padStart(2, '0');

      const m = String(now.getMinutes()).padStart(2, '0');

      document.getElementById('clock').textContent = h + ':' + m;

    }

    updateClock();

    setInterval(updateClock, 60000);


    // ========== Service Worker 注册 ==========

    if ('serviceWorker' in navigator) {

      window.addEventListener('load', () => {

        navigator.serviceWorker.register('sw.js').catch((err) => {

          console.log('Service Worker 注册失败:', err);

        });

      });

    }


    // ========== 通知权限 ==========

    if ('Notification' in window) {

      Notification.requestPermission();

    }


    // 初始渲染

    render();

  </script>
